This is the architecture of this project

ChaoticNotepad/
		├── backend/
		│   └── textengine/                              // pure C++17 editor engine, no Qt
		│       ├── dialog/
		│       │   ├── dialog.h                         // modal box with 2-3 buttons: which is selected and what each key does
		│       │   └── dialog.cpp                       // key, letter and click handling for that box
		│       ├── document/
		│       │   ├── document.h                       // the open file: path, line endings, encoding, open/save, detects changes on disk
		│       │   └── document.cpp                     // loading, saving, indent detection, disk change checks
		│       ├── filebrowser/
		│       │   ├── filebrowser.h                    // state of the Open / Save as panel: folder, filter, selection, keys, mouse
		│       │   └── filebrowser.cpp                  // lists folders, filters names, handles navigation and Enter
		│       ├── fileIO/
		│       │   ├── fileio.h                         // file helpers: paths, read, atomic write, encoding check, hashing
		│       │   └── fileio.cpp                       // their implementation
		│       ├── findpanel/
		│       │   ├── findpanel.h                      // find bar contents: buttons, match counter text, which toggles are on
		│       │   └── findpanel.cpp                    // builds the button list and counter text
		│       ├── findreplace/
		│       │   ├── findreplace.h                    // searches the text, keeps the match list, builds replacement text
		│       │   └── findreplace.cpp                  // plain and regex search, case folding, match navigation
		│       ├── folding/
		│       │   ├── folding.h                        // finds foldable blocks (brackets or indentation) and keeps fold state
		│       │   └── folding.cpp                      // region scanning, fold/unfold, hidden lines
		│       ├── gutter/
		│       │   ├── gutter.h                         // width and text of the line number column
		│       │   └── gutter.cpp                       // its implementation
		│       ├── gutterstyle/
		│       │   ├── gutterstyle.h                    // tracks which lines are unsaved, saved-changed or untouched
		│       │   └── gutterstyle.cpp                  // compares current text with opened and saved text
		│       ├── inputevents/
		│       │   └── inputevents.h                    // key, text, paste and mouse event types shared by all front ends
		│       ├── inputHandler/
		│       │   ├── inputhandler.h                   // input modes, requests to the app (save, open, tabs...), and the handleEvent entry points
		│       │   └── inputhandler.cpp                 // key bindings and editing, find and prompt behaviour
		│       ├── minimap/
		│       │   ├── minimap.h                        // computes the minimap picture, marks and scroll maths
		│       │   └── minimap.cpp                      // picture building and row/line conversion
		│       ├── multicursor/
		│       │   ├── multicursor.h                    // the Cursor type (caret + selection) and cursor merging
		│       │   └── multicursor.cpp                  // sorts and merges overlapping cursors
		│       ├── stickyscroll/
		│       │   ├── stickyscroll.h                   // decides which block header lines stay pinned at the top
		│       │   └── stickyscroll.cpp                 // scope tree and pinned line calculation
		│       ├── syntaxhighlight/
		│       │   ├── languagedef.h                    // the format of a language definition and the token kinds
		│       │   ├── syntaxhighlighter.h              // colours every character of the text by token kind
		│       │   ├── syntaxhighlighter.cpp            // line scanner with block comment and string state
		│       │   ├── cmakesyntax.h                    // CMake keywords and rules
		│       │   ├── cmakesyntax.cpp                  // its definition
		│       │   ├── cppsyntax.h                      // C/C++ keywords and rules
		│       │   ├── cppsyntax.cpp                    // its definition
		│       │   ├── fishsyntax.h                     // Fish shell keywords and rules
		│       │   ├── fishsyntax.cpp                   // its definition
		│       │   ├── jssyntax.h                       // JavaScript keywords and rules
		│       │   ├── jssyntax.cpp                     // its definition
		│       │   ├── luasyntax.h                      // Lua keywords and rules
		│       │   ├── luasyntax.cpp                    // its definition
		│       │   ├── qmlsyntax.h                      // QML keywords and rules
		│       │   ├── qmlsyntax.cpp                    // its definition
		│       │   ├── rustsyntax.h                     // Rust keywords and rules
		│       │   ├── rustsyntax.cpp                   // its definition
		│       │   ├── shsyntax.h                       // shell script keywords and rules
		│       │   └── shsyntax.cpp                     // its definition
		│       ├── tabbar/
		│       │   ├── tabbar.h                         // tab strip layout for a terminal front end (unused by QML)
		│       │   └── tabbar.cpp                       // columns, hit testing and drag reordering
		│       ├── tabs/
		│       │   ├── tabmanager.h                     // all open tabs, each with its view, document, gutter and syntax
		│       │   └── tabmanager.cpp                   // new, open, close, reopen, switch, move and reload tabs
		│       ├── textbuffer/
		│       │   ├── textbuffer.h                     // the text as a list of lines, positions and UTF-8 helpers
		│       │   └── textbuffer.cpp                   // insert, erase, offsets and clamping
		│       ├── textstats/
		│       │   ├── textstats.h                      // text counts: characters, words, lines, paragraphs, reading time
		│       │   └── textstats.cpp                    // the counting
		│       ├── textstyle/
		│       │   ├── textstyle.h                      // per-character bold, italic, underline and font size
		│       │   └── textstyle.cpp                    // keeps styles in step with the text
		│       ├── textview/
		│       │   ├── textview.h                       // the main editing class: text, cursors, undo, find, folds, indent
		│       │   └── textview.cpp                     // all editing, movement, find/replace and line operations
		│       ├── undohistory/
		│       │   ├── undohistory.h                    // undo and redo steps stored as small changes
		│       │   └── undohistory.cpp                  // step stack with size limits
		│       └── wordwrap/
		│           ├── wordwrap.h                       // splits lines into screen rows and converts positions to rows and back
		│           └── wordwrap.cpp                     // the wrap layout
		├── ChaoticNotepad.desktop                       // app launcher entry
		├── ChaoticNotepad.svg                           // app icon
		├── CMakeLists.txt                               // build instructions
		├── Features.txt                                 // feature list (still the 1.x text, needs updating)
		├── frontend/                                    // QML user interface
		│   ├── editor/
		│   │   ├── CodeEditor.qml                       // the main text area, scrolling and mouse input
		│   │   └── EditorRow.qml                        // one drawn line of text
		│   ├── filebrowserdialog/
		│   │   └── FileBrowserDialog.qml                // Open / Save as window
		│   ├── find/
		│   │   └── FindReplaceBar.qml                   // find and replace bar
		│   ├── fonts/
		│   │   └── fa-solid-900.ttf                     // icon font
		│   ├── gutter/
		│   │   ├── Gutter.qml                           // line number column
		│   │   └── GutterCell.qml                       // one line number or fold marker
		│   ├── Icon.qml                                 // icon characters used across the UI
		│   ├── minimap/
		│   │   └── Minimap.qml                          // whole-document overview on the right
		│   ├── promptdialog/
		│   │   └── PromptDialog.qml                     // "save changes?" prompt
		│   ├── settingsdialog/
		│   │   ├── FontListPanel.qml                    // searchable list of installed fonts
		│   │   ├── SettingsDialog.qml                   // settings window
		│   │   ├── SettingsRow.qml                      // one label + control row
		│   │   ├── SettingsStepper.qml                  // number +/- control
		│   │   └── SettingsSwitch.qml                   // on/off toggle
		│   ├── shortcutsdialog/
		│   │   └── ShortcutsDialog.qml                  // view and change keyboard shortcuts
		│   ├── statusbar/
		│   │   └── BottomStatusbar.qml                  // bottom bar: position, counts, indent, tabs
		│   ├── tabbar/
		│   │   ├── DocumentTabBar.qml                   // row of tabs with drag reordering
		│   │   └── DocumentTab.qml                      // one tab
		│   ├── Theme.qml                                // light and dark colours
		│   └── toolbar/
		│       ├── BottomToolbar.qml                    // button row (open, save, find, shortcuts, settings...)
		│       └── ToolbarButton.qml                    // one toolbar button
		├── importedcontent/
		│   └── README.md                                // old note (to delete)
		├── install.sh                                   // builds and installs system-wide into /usr/local (needs sudo)
		├── LICENSE                                      // license text
		├── main.cpp                                     // starts the app and opens files given on the command line
		├── Main.qml                                     // main window, puts all QML pieces together
		├── README.md                                    // project description
		├── tools/                                       // downloaded packaging tools (git-ignored)
		│   ├── appimagetool-x86_64.AppImage             // packs the AppImage
		│   ├── linuxdeploy-plugin-qt-x86_64.AppImage    // collects Qt for linuxdeploy
		│   └── linuxdeploy-x86_64.AppImage              // collects libraries into the AppDir
		├── tree.txt                                     // saved tree output (to delete)
		├── trial-build.log                              // old build log (to delete)
		├── uninstall.sh                                 // removes the install.sh install
		└── wrapper/                                     // Qt classes connecting the engine to QML
		    ├── browser/
		    │   ├── browsercontroller.h                  // file browser state and actions for QML
		    │   └── browsercontroller.cpp                // its implementation
		    ├── dialog/
		    │   ├── dialogcontroller.h                   // prompt dialog state and actions for QML
		    │   └── dialogcontroller.cpp                 // its implementation
		    ├── editor/
		    │   ├── editorcontroller.h                   // runs the engine: keys, mouse, scrolling, files, tabs
		    │   ├── editorcontroller.cpp                 // its implementation
		    │   ├── editorrowsmodel.h                    // list model of text rows for the editor view
		    │   ├── editorrowsmodel.cpp                  // its implementation
		    │   ├── rowbuilder.h                         // builds one row's text, colours and selection
		    │   └── rowbuilder.cpp                       // its implementation
		    ├── find/
		    │   ├── findcontroller.h                     // find/replace state and actions for QML
		    │   └── findcontroller.cpp                   // its implementation
		    ├── gutter/
		    │   ├── guttercontroller.h                   // gutter width and fold clicks for QML
		    │   ├── guttercontroller.cpp                 // its implementation
		    │   ├── guttermodel.h                        // list model of gutter rows
		    │   └── guttermodel.cpp                      // its implementation
		    ├── minimap/
		    │   ├── minimapcontroller.h                  // minimap position, clicks and dragging for QML
		    │   ├── minimapcontroller.cpp                // its implementation
		    │   ├── minimappicture.h                     // QML item that draws the minimap image
		    │   └── minimappicture.cpp                   // its implementation
		    ├── settings/
		    │   ├── settingscontroller.h                 // font, theme, indent and other settings for QML
		    │   └── settingscontroller.cpp               // its implementation, saved between runs
		    ├── shortcuts/
		    │   ├── shortcutscontroller.h                // editable shortcut list for QML
		    │   └── shortcutscontroller.cpp              // its implementation, saved between runs
		    ├── status/
		    │   ├── statuscontroller.h                   // status bar values for QML
		    │   └── statuscontroller.cpp                 // its implementation
		    └── tabs/
		        ├── tabscontroller.h                     // tab list model and tab actions for QML
		        └── tabscontroller.cpp                   // its implementation
